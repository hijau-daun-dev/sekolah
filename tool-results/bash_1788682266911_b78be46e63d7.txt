    98→// ============ Generate Kelas (Kenaikan) ============
    99→function GenerateKelasCard() {
   100→  const [taList, setTaList] = useState<TahunAjaran[]>([]);
   101→  const [open, setOpen] = useState(false);
   102→  const [lamaId, setLamaId] = useState<number | null>(null);
   103→  const [baruId, setBaruId] = useState<number | null>(null);
   104→  const [submitting, setSubmitting] = useState(false);
   105→  const [result, setResult] = useState<null | { promoted: number; graduated: number; skipped: number; total: number }>(null);
   106→  const { toast } = useToast();
   107→
   108→  useEffect(() => {
   109→    fetch("/api/tahun-ajaran")
   110→      .then((r) => r.json())
   111→      .then((d: TahunAjaran[]) => {
   112→        if (Array.isArray(d)) setTaList(d);
   113→      })
   114→      .catch(() => { /* ignore */ });
   115→  }, []);
   116→
   117→  const handleOpen = () => {
   118→    setResult(null);
   119→    setLamaId(null);
   120→    setBaruId(null);
   121→    setOpen(true);
   122→  };
   123→
   124→  const handleSubmit = async () => {
   125→    if (!lamaId || !baruId) {
   126→      toast({ title: "Pilih tahun ajaran lama dan baru", variant: "destructive" });
   127→      return;
   128→    }
   129→    if (lamaId === baruId) {
   130→      toast({ title: "Tahun ajaran lama dan baru tidak boleh sama", variant: "destructive" });
   131→      return;
   132→    }
   133→    setSubmitting(true);
   134→    try {
   135→      const r = await fetch("/api/generate-kelas", {
   136→        method: "POST",
   137→        headers: { "Content-Type": "application/json" },
   138→        body: JSON.stringify({ tahunAjaranIdLama: lamaId, tahunAjaranIdBaru: baruId }),
   139→      });
   140→      const d = await r.json();
   141→      if (!r.ok) throw new Error(d.error || "Gagal generate kelas");
   142→      setResult({ promoted: d.promoted, graduated: d.graduated, skipped: d.skipped, total: d.total });
   143→      toast({
   144→        title: "Generate Kelas selesai",
   145→        description: `Naik: ${d.promoted} • Lulus: ${d.graduated} • Skip: ${d.skipped} (total: ${d.total})`,
   146→      });
   147→    } catch (e) {
   148→      toast({
   149→        title: "Gagal generate kelas",
   150→        description: e instanceof Error ? e.message : "",
   151→        variant: "destructive",
   152→      });
   153→    } finally {
   154→      setSubmitting(false);
   155→    }
   156→  };
   157→
   158→  return (
   159→    <Card className="border-slate-200">
   160→      <CardContent className="p-4">
   161→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   162→          <div>
   163→            <h3 className="text-base font-semibold text-slate-800 flex items-center gap-2">
   164→              <ArrowUpRight className="h-4 w-4 text-slate-600" />
   165→              Generate Kelas (Kenaikan Kelas)
   166→            </h3>
   167→            <p className="text-xs text-slate-500 mt-0.5">
   168→              Naikkan siswa ke kelas di tahun ajaran baru sesuai tingkat berikutnya. Siswa di kelas akhir akan ditandai sebagai &quot;Lulus&quot;.
   169→            </p>
   170→          </div>
   171→          <Button size="sm" onClick={handleOpen} className="bg-slate-700 hover:bg-slate-800 whitespace-nowrap">
   172→            <ArrowUpRight className="h-4 w-4 mr-1" /> Generate Kelas
   173→          </Button>
   174→        </div>
   175→      </CardContent>
   176→
   177→      <Dialog open={open} onOpenChange={(o) => { setOpen(o); if (!o) setResult(null); }}>
   178→        <DialogContent className="max-w-lg">
   179→          <DialogHeader>
   180→            <DialogTitle>Generate Kelas (Kenaikan)</DialogTitle>
   181→            <DialogDescription>
   182→              Pilih tahun ajaran lama sebagai sumber, dan tahun ajaran baru sebagai tujuan kenaikan.
   183→            </DialogDescription>
   184→          </DialogHeader>
   185→          <div className="grid grid-cols-1 gap-3 py-2">
   186→            <div>
   187→              <Label className="text-sm">Tahun Ajaran Lama (Sumber) *</Label>
   188→              <Select
   189→                value={lamaId ? String(lamaId) : "none"}
   190→                onValueChange={(v) => setLamaId(v === "none" ? null : Number(v))}
   191→              >
   192→                <SelectTrigger className="w-full"><SelectValue placeholder="Pilih tahun ajaran lama" /></SelectTrigger>
   193→                <SelectContent>
   194→                  <SelectItem value="none" disabled>— Pilih —</SelectItem>
   195→                  {taList.map((t) => (
   196→                    <SelectItem key={t.id} value={String(t.id)}>
   197→                      {t.nama}{t.statusAktif ? " (Aktif)" : ""}
   198→                    </SelectItem>
   199→                  ))}
   200→                </SelectContent>
   201→              </Select>
   202→            </div>
   203→            <div>
   204→              <Label className="text-sm">Tahun Ajaran Baru (Tujuan) *</Label>
   205→              <Select
   206→                value={baruId ? String(baruId) : "none"}
   207→                onValueChange={(v) => setBaruId(v === "none" ? null : Number(v))}
   208→              >
   209→                <SelectTrigger className="w-full"><SelectValue placeholder="Pilih tahun ajaran baru" /></SelectTrigger>
   210→                <SelectContent>
   211→                  <SelectItem value="none" disabled>— Pilih —</SelectItem>
   212→                  {taList.map((t) => (
   213→                    <SelectItem key={t.id} value={String(t.id)}>
   214→                      {t.nama}{t.statusAktif ? " (Aktif)" : ""}
   215→                    </SelectItem>
   216→                  ))}
   217→                </SelectContent>
   218→              </Select>
   219→            </div>
   220→            <p className="text-xs text-slate-500 leading-relaxed">
   221→              Catatan: pastikan kelas pada tahun ajaran baru sudah dibuat (menu Kelas &amp; Siswa) dengan
   222→              nama mengikuti pola tingkat (cth. &quot;6A&quot; untuk tingkat 6 paralel A). Siswa yang tidak
   223→              memiliki kelas tujuan akan dilewati (skip).
   224→            </p>
   225→
   226→            {result && (
   227→              <div className="rounded-md border border-slate-200 bg-slate-50 p-3 text-sm space-y-1">
   228→                <div className="font-medium text-slate-800">Hasil:</div>
   229→                <div className="grid grid-cols-2 gap-x-3 gap-y-1 text-xs text-slate-700">
   230→                  <span> Total diproses:</span><span className="text-right font-semibold">{result.total}</span>
   231→                  <span> Naik kelas:</span><span className="text-right font-semibold text-emerald-700">{result.promoted}</span>
   232→                  <span> Lulus:</span><span className="text-right font-semibold text-amber-700">{result.graduated}</span>
   233→                  <span> Dilewati (skip):</span><span className="text-right font-semibold text-slate-600">{result.skipped}</span>
   234→                </div>
   235→              </div>
   236→            )}
   237→          </div>
   238→          <DialogFooter>
   239→            <Button variant="outline" onClick={() => setOpen(false)} disabled={submitting}>
   240→              {result ? "Tutup" : "Batal"}
   241→            </Button>
   242→            <Button onClick={handleSubmit} disabled={submitting} className="bg-slate-700 hover:bg-slate-800">
   243→              {submitting && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   244→              Jalankan Generate
   245→            </Button>
   246→          </DialogFooter>
   247→        </DialogContent>
   248→      </Dialog>
   249→    </Card>
   250→  );
   251→}

   382→// ============ Kelas (with kelas-siswa management) ============
   383→function KelasTab() {
   384→  const [tingkatList, setTingkatList] = useState<{ value: string; label: string }[]>([]);
   385→  const [jurusanList, setJurusanList] = useState<{ value: string; label: string }[]>([]);
   386→  const [taList, setTaList] = useState<{ value: string; label: string }[]>([]);
   387→  const [pegawaiList, setPegawaiList] = useState<{ value: string; label: string }[]>([]);
   388→  const [list, setList] = useState<Kelas[]>([]);
   389→  const [loading, setLoading] = useState(true);
   390→  const [search, setSearch] = useState("");
   391→  const [dialogOpen, setDialogOpen] = useState(false);
   392→  const [editing, setEditing] = useState<Partial<Kelas> | null>(null);
   393→  const [saving, setSaving] = useState(false);
   394→  const [delTarget, setDelTarget] = useState<Kelas | null>(null);
   395→  const [siswaDialogKelas, setSiswaDialogKelas] = useState<Kelas | null>(null);
   396→  const { toast } = useToast();
   397→
   398→  const load = useCallback(async () => {
   399→    setLoading(true);
   400→    try {
   401→      const r = await fetch("/api/kelas");
   402→      const d = await r.json();
   403→      if (Array.isArray(d)) setList(d);
   404→    } catch {
   405→      toast({ title: "Gagal memuat kelas", variant: "destructive" });
   406→    } finally {
   407→      setLoading(false);
   408→    }
   409→  }, [toast]);
   410→
   411→  useEffect(() => {
   412→    load();
   413→    Promise.all([
   414→      fetch("/api/tingkat").then((r) => r.json()),

   640→function KelasSiswaManager({ kelas }: { kelas: Kelas }) {
   641→  const [list, setList] = useState<KelasSiswa[]>([]);
   642→  const [allSiswa, setAllSiswa] = useState<SiswaMini[]>([]);
   643→  const [loading, setLoading] = useState(true);
   644→  const [pick, setPick] = useState<string>("");
   645→  const [adding, setAdding] = useState(false);
   646→  const { toast } = useToast();
   647→
   648→  const load = useCallback(async () => {
   649→    setLoading(true);
   650→    try {
   651→      const [r1, r2] = await Promise.all([
   652→        fetch(`/api/kelas-siswa?kelasId=${kelas.id}`),
   653→        fetch("/api/siswa"),
   654→      ]);
   655→      const d1 = await r1.json();
   656→      const d2 = await r2.json();
   657→      if (Array.isArray(d1)) setList(d1);
   658→      if (Array.isArray(d2)) setAllSiswa(d2);
   659→    } catch { /* ignore */ } finally { setLoading(false); }
   660→  }, [kelas.id]);
   661→
   662→  useEffect(() => { load(); }, [load]);
   663→
   664→  const linkedIds = new Set(list.map((x) => x.siswaId));
   665→  const available = allSiswa.filter((s) => !linkedIds.has(s.id));
   666→
   667→  const handleAdd = async () => {
   668→    if (!pick) return;
   669→    setAdding(true);
   670→    try {
   671→      const r = await fetch("/api/kelas-siswa", {
   672→        method: "POST", headers: { "Content-Type": "application/json" },
   673→        body: JSON.stringify({ kelasId: kelas.id, siswaId: Number(pick), tahunAjaranId: kelas.tahunAjaranId }),
   674→      });
   675→      const d = await r.json();
   676→      if (!r.ok) throw new Error(d.error || "Gagal menambah");
   677→      setPick("");
   678→      await load();
   679→      toast({ title: "Siswa ditambahkan ke kelas" });
   680→    } catch (e) {
   681→      toast({ title: "Gagal menambah", description: e instanceof Error ? e.message : "", variant: "destructive" });
   682→    } finally { setAdding(false); }
   683→  };
   684→
   685→  const handleRemove = async (ksid: number, nama: string) => {
   686→    try {
   687→      const r = await fetch(`/api/kelas-siswa/${ksid}`, { method: "DELETE" });
   688→      if (!r.ok) throw new Error("Gagal menghapus");
   689→      await load();
   690→      toast({ title: `${nama} dihapus dari kelas` });
   691→    } catch (e) {
   692→      toast({ title: "Gagal menghapus", description: e instanceof Error ? e.message : "", variant: "destructive" });
   693→    }
   694→  };
   695→
   696→  return (
   697→    <div className="space-y-3">
   698→      {loading ? (
   699→        <div className="flex justify-center py-6"><Loader2 className="h-5 w-5 animate-spin text-slate-500" /></div>
   700→      ) : (
   701→        <>
   702→          <div className="flex flex-col sm:flex-row gap-2 items-end">
   703→            <div className="flex-1 w-full">
   704→              <Label className="text-xs">Tambah Siswa ke Kelas</Label>
   705→              <Select value={pick} onValueChange={setPick}>
   706→                <SelectTrigger className="w-full"><SelectValue placeholder="Pilih siswa..." /></SelectTrigger>
   707→                <SelectContent>
   708→                  {available.length === 0 ? (
   709→                    <SelectItem value="_none" disabled>Tidak ada siswa tersedia</SelectItem>
   710→                  ) : available.map((s) => (
   711→                    <SelectItem key={s.id} value={String(s.id)}>{s.nama} {s.nis ? `(${s.nis})` : ""}</SelectItem>
   712→                  ))}
   713→                </SelectContent>
   714→              </Select>
   715→            </div>
   716→            <Button onClick={handleAdd} disabled={adding || !pick || pick === "_none"} className="bg-slate-700 hover:bg-slate-800">
   717→              {adding ? <Loader2 className="h-4 w-4 animate-spin" /> : <Plus className="h-4 w-4 mr-1" />} Tambah
   718→            </Button>
   719→          </div>
   720→          <div className="border border-slate-200 rounded-lg overflow-hidden max-h-72 overflow-y-auto">
   721→            {list.length === 0 ? (
   722→              <div className="text-center py-6 text-sm text-slate-500">Belum ada siswa di kelas ini.</div>
   723→            ) : (
   724→              <table className="w-full text-sm">
   725→                <thead className="bg-slate-50 text-slate-600 sticky top-0">
   726→                  <tr>
   727→                    <th className="text-left px-3 py-2 font-medium">Nama</th>
   728→                    <th className="text-left px-3 py-2 font-medium">NIS</th>
   729→                    <th className="text-left px-3 py-2 font-medium">Status</th>
   730→                    <th className="text-right px-3 py-2 font-medium w-12"></th>
   731→                  </tr>
   732→                </thead>
   733→                <tbody className="divide-y divide-slate-100">
   734→                  {list.map((ks) => (
   735→                    <tr key={ks.id} className="hover:bg-slate-50/60">
   736→                      <td className="px-3 py-2 font-medium text-slate-800">{ks.siswa?.nama || "-"}</td>
   737→                      <td className="px-3 py-2 text-xs">{ks.siswa?.nis || "-"}</td>
   738→                      <td className="px-3 py-2"><Badge variant="outline" className="text-[10px]">{ks.siswa?.status || "-"}</Badge></td>
   739→                      <td className="px-3 py-2 text-right">

   741→                          <X className="h-3.5 w-3.5" />
   742→                        </Button>
   743→                      </td>
   744→                    </tr>

   746→                </tbody>
   747→              </table>
   748→            )}
   749→          </div>

   848→// ============ Guru Mapel ============
   849→function GuruMapelTab() {
   850→  const [pgList, setPgList] = useState<{ value: string; label: string }[]>([]);
   851→  const pList, setMpList] = useState<{ value: string; label: string }[]>([]);
   852→  const [kList, setKList] = useState<{ value: string; label: string }[]>([]);
   853→  const [list, setList] = useState<GuruMapel[]>([]);
   854→  const [loading, setLoading] = useState(true);
   855→  const [dialogOpen, setDialogOpen] = useState(false);
   856→  const [form, setForm] = useState<{ pegawaiId: string; mapelId: string; kelasId: string }>({ pegawaiId: "", mapelId: "", kelasId: "" });
   857→  const [saving, setSaving] = useState(false);
   858→  const [delTarget, setDelTarget] = useState<GuruMapel | null>(null);
   859→  const { toast } = useToast();
   860→
   861→  const load = useCallback(async () => {
   862→    setLoading(true);
   863→    try {
   864→      const r = await fetch("/api/guru-mapel");
   865→      const d = await r.json();
   866→      if (Array.isArray(d)) setList(d);
   867→    } catch { /* ignore */ } finally { setLoading(false); }
   868→  }, []);
   869→
   870→  useEffect(() => {
   871→    load();
   872→    Promise.all([
   873→      fetch("/api/pegawai").then((r) => r.json()),
   874→      fetch("/api/mapel").then((r) => r.json()),
   875→      fetch("/api/kelas").then((r) => r.json()),
   876→    ]).then(([pg, mp, kl]: [PegawaiMini[], Mapel[], Kelas[]]) => {
   877→      if (Array.isArray(pg)) setPgList(pg.map((x) => ({ value: String(x.id), label: x.nama + (x.jabatan ? ` (${x.jabatan})` : "") })));
   878→      if (Array.isArray(mp)) setMpList(mp.map((x) => ({ value: String(x.id), label: x.nama + (x.kode ? ` [${x.kode}]` : "") })));
   879→      if (Array.isArray(kl)) setKList(kl.map((x) => ({ value: String(x.id), label: x.nama })));
   880→    });
   881→  }, [load]);
   882→
   883→  const handleAdd = async () => {
   884→    if (!form.pegawaiId || !form.mapelId) { toast({ title: "Pegawai & Mapel wajib", variant: "destructive" }); return; }
   885→    setSaving(true);
   886→    try {
   887→      const r = await fetch("/api/guru-mapel", {
   888→        method: "POST", headers: { "Content-Type": "application/json" },
   889→        body: JSON.stringify({
   890→          pegawaiId: Number(form.pegawaiId),
   891→          mapelId: Number(form.mapelId),
   892→          kelasId: form.kelasId ? Number(form.kelasId) : null,
   893→        }),
   894→      });
   895→      const d = await r.json();
   896→      if (!r.ok) throw new Error(d.error || "Gagal menambah");
   897→      toast({ title: "Guru mapel ditambahkan" });
   898→      setDialogOpen(false);
   899→      setForm({ pegawaiId: "", mapelId: "", kelasId: "" });
   900→      await load();
   901→    } catch (e) {
   902→      toast({ title: "Gagal menambah", description: e instanceof Error ? e.message : "", variant: "destructive" });
   903→    } finally { setSaving(false); }
   904→  };
   905→
   906→  const handleDelete = async () => {
   907→    if (!delTarget) return;
   908→    try {
   909→      const r = await fetch(`/api/guru-mapel/${delTarget.id}`, { method: "DELETE" });
   910→      if (!r.ok) throw new Error("Gagal menghapus");
   911→      toast({ title: "Guru mapel dihapus" });
   912→      setDelTarget(null);
   913→      await load();
   914→    } catch (e) {
   915→      toast({ title: "Gagal menghapus", description: e instanceof Error ? e.message : "", variant: "destructive" });
   916→    }
   917→  };
   918→
   919→  return (
   920→    <Card className="border-slate-200">
   921→      <CardContent className="p-4 space-y-4">
   922→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   923→          <div>
   924→            <h3 className="text-base font-semibold text-slate-800">Guru Mapel</h3>
   925→            <p className="text-xs text-slate-500">Pasangan guru – mapel – kelas</p>
   926→          </div>
   927→          <Button size="sm" onClick={() => setDialogOpen(true)} className="bg-slate-700 hover:bg-slate-800"><Plus className="h-4 w-4 mr-1" /> Tambah</Button>
   928→        </div>
   929→        {loading ? (
   930→          <div className="flex justify-center py-8"><Loader2 className="h-5 w-5 animate-spin text-slate-500" /></div>
   931→        ) : list.length === 0 ? (
   932→          <div className="text-center py-8 text-slate-500"><p className="text-sm">Belum ada data.</p></div>
   933→        ) : (
   934→          <div className="border border-slate-200 rounded-lg overflow-x-auto">
   935→            <table className="w-full text-sm">
   936→              <thead className="bg-slate-50 text-slate-600">
   937→                <tr>
   938→                  <th className="text-left px-3 py-2 font-medium">Guru</th>
   939→                  <th className="text-left px-3 py-2 font-medium">Mapel</th>
   940→                  <th className="text-left px-3 py-2 font-medium">Kelas</th>
   941→                  <th className="text-right px-3 py-2 font-medium w-12"></th>
   942→                </tr>
   943→              </thead>
   944→              <tbody className="divide-y divide-slate-100">
   945→                {list.map((g) => (
   946→                  <tr key={g.id} className="hover:bg-slate-50/60">
   947→                    <td className="px-3 py-2 font-medium text-slate-800">{g.pegawai?.nama || "-"}</td>
   948→                    <td className="px-3 py-2">{g.mapel?.nama || "-"}</td>
   949→                    <td className="px-3 py-2">{g.kelas?.nama || <span className="text-slate-400">Semua</span>}</td>
   950→                    <td className="px-3 py-2 text-right">
   951→                      <Button size="icon" variant="ghost" className="h-7 w-7 text-rose-600 hover:bg-rose-50" onClick={() => setDelTarget(g)}>
   952→                        <Trash2 className="h-3.5 w-3.5" />
   953→                      </Button>
   954→                    </td>
   955→                  </tr>
   956→                ))}
   957→              </tbody>
   958→            </table>
   959→          </div>
   960→        )}
   961→      </CardContent>
   962→
   963→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   964→        <DialogContent className="max-w-md">
   965→          <DialogHeader>
   966→            <DialogTitle>Tambah Guru Mapel</DialogTitle>
   967→            <DialogDescription>Tetapkan guru untuk mapel & kelas tertentu.</DialogDescription>
   968→          </DialogHeader>
   969→          <div className="space-y-3 py-2">
   970→            <div>
   971→              <Label className="text-sm">Guru / Pegawai *</Label>
   972→              <Select value={form.pegawaiId} onValueChange={(v) => setForm((p) => ({ ...p, pegawaiId: v }))}>
   973→                <SelectTrigger className="w-full"><SelectValue placeholder="Pilih guru" /></SelectTrigger>
   974→                <SelectContent>
   975→                  {pgList.map((o) => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}
   976→                </SelectContent>
   977→              </Select>
   978→            </div>
   979→            <div>
   980→              <Label className="text-sm">Mata Pelajaran *</Label>
   981→              <Select value={form.mapelId} onValueChange={(v) => setForm((p) => ({ ...p, mapelId: v }))}>
   982→                <SelectTrigger className="w-full"><SelectValue placeholder="Pilih mapel" /></SelectTrigger>
   983→                <SelectContent>
   984→                  {mpList.map((o) => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}
   985→                </SelectContent>
   986→              </Select>
   987→            </div>
   988→            <div>
   989→              <Label className="text-sm">Kelas (opsional)</Label>
   990→              <Select value={form.kelasId || "none"} onValueChange={(v) => setForm((p) => ({ ...p, kelasId: v === "none" ? "" : v }))}>
   991→                <SelectTrigger className="w-full"><SelectValue placeholder="Semua kelas" /></SelectTrigger>
   992→                <SelectContent>
   993→                  <SelectItem value="none">— Semua Kelas —</SelectItem>
   994→                  {kList.map((o) => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}
   995→                </SelectContent>
   996→              </Select>
   997→            </div>
   998→          </div>
   999→          <DialogFooter>
  1000→            <Button variant="outline" onClick={() => setDialogOpen(false)}>Batal</Button>
  1001→            <Button onClick={handleAdd} disabled={saving} className="bg-slate-700 hover:bg-slate-800">
  1002→              {saving && <Loader2 className="h-4 w-4 mr-2 animate-spin" />} Simpan
  1003→            </Button>
  1004→          </DialogFooter>
  1005→        </DialogContent>
  1006→      </Dialog>
  1007→
  1008→      <Dialog open={!!delTarget} onOpenChange={(o) => !o && setDelTarget(null)}>
  1009→        <DialogContent className="max-w-md">
  1010→          <DialogHeader>
  1011→            <DialogTitle>Hapus guru mapel?</DialogTitle>
  1012→            <DialogDescription>Yakin menghapus penugasan guru ini?</DialogDescription>
  1013→          </DialogHeader>
  1014→          <DialogFooter>
  1015→            <Button variant="outline" onClick={() => setDelTarget(null)}>Batal</Button>
  1016→            <Button onClick={handleDelete} className="bg-rose-600 hover:bg-rose-700">Hapus</Button>
  1017→          </DialogFooter>
  1018→        </DialogContent>
  1019→      </Dialog>
  1020→    </Card>
  1021→  );
  1022→}
