    98→    98→// ============ Generate Kelas (Kenaikan) ============
    99→    99→function GenerateKelasCard() {
   100→   100→  const [taList, setTaList] = useState<TahunAjaran[]>([]);
   101→   101→  const [open, setOpen] = useState(false);
   102→   102→  const [lamaId, setLamaId] = useState<number | null>(null);
   103→   103→  const [baruId, setBaruId] = useState<number | null>(null);
   104→   104→  const [submitting, setSubmitting] = useState(false);
   105→   105→  const [result, setResult] = useState<null | { promoted: number; graduated: number; skipped: number; total: number }>(null);
   106→   106→  const { toast } = useToast();
   107→   107→
   108→   108→  useEffect(() => {
   109→   109→    fetch("/api/tahun-ajaran")
   110→   110→      .then((r) => r.json())
   111→   111→      .then((d: TahunAjaran[]) => {
   112→   112→        if (Array.isArray(d)) setTaList(d);
   113→   113→      })
   114→   114→      .catch(() => { /* ignore */ });
   115→   115→  }, []);
   116→   116→
   117→   117→  const handleOpen = () => {
   118→   118→    setResult(null);
   119→   119→    setLamaId(null);
   120→   120→    setBaruId(null);
   121→   121→    setOpen(true);
   122→   122→  };
   123→   123→
   124→   124→  const handleSubmit = async () => {
   125→   125→    if (!lamaId || !baruId) {
   126→   126→      toast({ title: "Pilih tahun ajaran lama dan baru", variant: "destructive" });
   127→   127→      return;
   128→   128→    }
   129→   129→    if (lamaId === baruId) {
   130→   130→      toast({ title: "Tahun ajaran lama dan baru tidak boleh sama", variant: "destructive" });
   131→   131→      return;
   132→   132→    }
   133→   133→    setSubmitting(true);
   134→   134→    try {
   135→   135→      const r = await fetch("/api/generate-kelas", {
   136→   136→        method: "POST",
   137→   137→        headers: { "Content-Type": "application/json" },
   138→   138→        body: JSON.stringify({ tahunAjaranIdLama: lamaId, tahunAjaranIdBaru: baruId }),
   139→   139→      });
   140→   140→      const d = await r.json();
   141→   141→      if (!r.ok) throw new Error(d.error || "Gagal generate kelas");
   142→   142→      setResult({ promoted: d.promoted, graduated: d.graduated, skipped: d.skipped, total: d.total });
   143→   143→      toast({
   144→   144→        title: "Generate Kelas selesai",
   145→   145→        description: `Naik: ${d.promoted} • Lulus: ${d.graduated} • Skip: ${d.skipped} (total: ${d.total})`,
   146→   146→      });
   147→   147→    } catch (e) {
   148→   148→      toast({
   149→   149→        title: "Gagal generate kelas",
   150→   150→        description: e instanceof Error ? e.message : "",
   151→   151→        variant: "destructive",
   152→   152→      });
   153→   153→    } finally {
   154→   154→      setSubmitting(false);
   155→   155→    }
   156→   156→  };
   157→   157→
   158→   158→  return (
   159→   159→    <Card className="border-slate-200">
   160→   160→      <CardContent className="p-4">
   161→   161→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   162→   162→          <div>
   163→   163→            <h3 className="text-base font-semibold text-slate-800 flex items-center gap-2">
   164→   164→              <ArrowUpRight className="h-4 w-4 text-slate-600" />
   165→   165→              Generate Kelas (Kenaikan Kelas)
   166→   166→            </h3>
   167→   167→            <p className="text-xs text-slate-500 mt-0.5">
   168→   168→              Naikkan siswa ke kelas di tahun ajaran baru sesuai tingkat berikutnya. Siswa di kelas akhir akan ditandai sebagai &quot;Lulus&quot;.
   169→   169→            </p>
   170→   170→          </div>
   171→   171→          <Button size="sm" onClick={handleOpen} className="bg-slate-700 hover:bg-slate-800 whitespace-nowrap">
   172→   172→            <ArrowUpRight className="h-4 w-4 mr-1" /> Generate Kelas
   173→   173→          </Button>
   174→   174→        </div>
   175→   175→      </CardContent>
   176→   176→
   177→   177→      <Dialog open={open} onOpenChange={(o) => { setOpen(o); if (!o) setResult(null); }}>
   178→   178→        <DialogContent className="max-w-lg">
   179→   179→          <DialogHeader>
   180→   180→            <DialogTitle>Generate Kelas (Kenaikan)</DialogTitle>
   181→   181→            <DialogDescription>
   182→   182→              Pilih tahun ajaran lama sebagai sumber, dan tahun ajaran baru sebagai tujuan kenaikan.
   183→   183→            </DialogDescription>
   184→   184→          </DialogHeader>
   185→   185→          <div className="grid grid-cols-1 gap-3 py-2">
   186→   186→            <div>
   187→   187→              <Label className="text-sm">Tahun Ajaran Lama (Sumber) *</Label>
   188→   188→              <Select
   189→   189→                value={lamaId ? String(lamaId) : "none"}
   190→   190→                onValueChange={(v) => setLamaId(v === "none" ? null : Number(v))}
   191→   191→              >
   192→   192→                <SelectTrigger className="w-full"><SelectValue placeholder="Pilih tahun ajaran lama" /></SelectTrigger>
   193→   193→                <SelectContent>
   194→   194→                  <SelectItem value="none" disabled>— Pilih —</SelectItem>
   195→   195→                  {taList.map((t) => (
   196→   196→                    <SelectItem key={t.id} value={String(t.id)}>
   197→   197→                      {t.nama}{t.statusAktif ? " (Aktif)" : ""}
   198→   198→                    </SelectItem>
   199→   199→                  ))}
   200→   200→                </SelectContent>
   201→   201→              </Select>
   202→   202→            </div>
   203→   203→            <div>
   204→   204→              <Label className="text-sm">Tahun Ajaran Baru (Tujuan) *</Label>
   205→   205→              <Select
   206→   206→                value={baruId ? String(baruId) : "none"}
   207→   207→                onValueChange={(v) => setBaruId(v === "none" ? null : Number(v))}
   208→   208→              >
   209→   209→                <SelectTrigger className="w-full"><SelectValue placeholder="Pilih tahun ajaran baru" /></SelectTrigger>
   210→   210→                <SelectContent>
   211→   211→                  <SelectItem value="none" disabled>— Pilih —</SelectItem>
   212→   212→                  {taList.map((t) => (
   213→   213→                    <SelectItem key={t.id} value={String(t.id)}>
   214→   214→                      {t.nama}{t.statusAktif ? " (Aktif)" : ""}
   215→   215→                    </SelectItem>
   216→   216→                  ))}
   217→   217→                </SelectContent>
   218→   218→              </Select>
   219→   219→            </div>
   220→   220→            <p className="text-xs text-slate-500 leading-relaxed">
   221→   221→              Catatan: pastikan kelas pada tahun ajaran baru sudah dibuat (menu Kelas &amp; Siswa) dengan
   222→   222→              nama mengikuti pola tingkat (cth. &quot;6A&quot; untuk tingkat 6 paralel A). Siswa yang tidak
   223→   223→              memiliki kelas tujuan akan dilewati (skip).
   224→   224→            </p>
   225→   225→
   226→   226→            {result && (
   227→   227→              <div className="rounded-md border border-slate-200 bg-slate-50 p-3 text-sm space-y-1">
   228→   228→                <div className="font-medium text-slate-800">Hasil:</div>
   229→   229→                <div className="grid grid-cols-2 gap-x-3 gap-y-1 text-xs text-slate-700">
   230→   230→                  <span> Total diproses:</span><span className="text-right font-semibold">{result.total}</span>
   231→   231→                  <span> Naik kelas:</span><span className="text-right font-semibold text-emerald-700">{result.promoted}</span>
   232→   232→                  <span> Lulus:</span><span className="text-right font-semibold text-amber-700">{result.graduated}</span>
   233→   233→                  <span> Dilewati (skip):</span><span className="text-right font-semibold text-slate-600">{result.skipped}</span>
   234→   234→                </div>
   235→   235→              </div>
   236→   236→            )}
   237→   237→          </div>
   238→   238→          <DialogFooter>
   239→   239→            <Button variant="outline" onClick={() => setOpen(false)} disabled={submitting}>
   240→   240→              {result ? "Tutup" : "Batal"}
   241→   241→            </Button>
   242→   242→            <Button onClick={handleSubmit} disabled={submitting} className="bg-slate-700 hover:bg-slate-800">
   243→   243→              {submitting && <Loader2 className="h-4 w-4 mr-2 animate-spin" />}
   244→   244→              Jalankan Generate
   245→   245→            </Button>
   246→   246→          </DialogFooter>
   247→   247→        </DialogContent>
   248→   248→      </Dialog>
   249→   249→    </Card>
   250→   250→  );
   251→   251→}
