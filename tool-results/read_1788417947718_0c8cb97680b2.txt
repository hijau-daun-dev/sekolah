   226→// ============ Kelas (with kelas-siswa management) ============
   227→function KelasTab() {
   228→  const [tingkatList, setTingkatList] = useState<{ value: string; label: string }[]>([]);
   229→  const [jurusanList, setJurusanList] = useState<{ value: string; label: string }[]>([]);
   230→  const [taList, setTaList] = useState<{ value: string; label: string }[]>([]);
   231→  const [pegawaiList, setPegawaiList] = useState<{ value: string; label: string }[]>([]);
   232→  const [list, setList] = useState<Kelas[]>([]);
   233→  const [loading, setLoading] = useState(true);
   234→  const [search, setSearch] = useState("");
   235→  const [dialogOpen, setDialogOpen] = useState(false);
   236→  const [editing, setEditing] = useState<Partial<Kelas> | null>(null);
   237→  const [saving, setSaving] = useState(false);
   238→  const [delTarget, setDelTarget] = useState<Kelas | null>(null);
   239→  const [siswaDialogKelas, setSiswaDialogKelas] = useState<Kelas | null>(null);
   240→  const { toast } = useToast();
   241→
   242→  const load = useCallback(async () => {
   243→    setLoading(true);
   244→    try {
   245→      const r = await fetch("/api/kelas");
   246→      const d = await r.json();
   247→      if (Array.isArray(d)) setList(d);
   248→    } catch {
   249→      toast({ title: "Gagal memuat kelas", variant: "destructive" });
   250→    } finally {
   251→      setLoading(false);
   252→    }
   253→  }, [toast]);
   254→
   255→  useEffect(() => {
   256→    load();
   257→    Promise.all([
   258→      fetch("/api/tingkat").then((r) => r.json()),
   259→      fetch("/api/jurusan").then((r) => r.json()),
   260→      fetch("/api/tahun-ajaran").then((r) => r.json()),
   261→      fetch("/api/pegawai").then((r) => r.json()),
   262→    ]).then(([t, j, ta, pg]: [Tingkat[], Jurusan[], TahunAjaran[], PegawaiMini[]]) => {
   263→      if (Array.isArray(t)) setTingkatList(t.map((x) => ({ value: String(x.id), label: `${x.nama}${x.jenjang ? ` (${x.jenjang})` : ""}` })));
   264→      if (Array.isArray(j)) setJurusanList(j.map((x) => ({ value: String(x.id), label: `${x.kode} - ${x.nama}` })));
   265→      if (Array.isArray(ta)) setTaList(ta.map((x) => ({ value: String(x.id), label: x.nama + (x.statusAktif ? " (Aktif)" : "") })));
   266→      if (Array.isArray(pg)) setPegawaiList(pg.map((x) => ({ value: String(x.id), label: x.nama + (x.jabatan ? ` (${x.jabatan})` : "") })));
   267→    });
   268→  }, [load]);
   269→
   270→  const filtered = list.filter((k) => {
   271→    if (!search) return true;
   272→    const q = search.toLowerCase();
   273→    return k.nama.toLowerCase().includes(q) || (k.tingkat?.nama || "").toLowerCase().includes(q);
   274→  });
   275→
   276→  const handleAdd = () => {
   277→    // prefill tahun ajaran aktif
   278→    const aktifTa = taList.length > 0 ? taList[0].value : "";
   279→    setEditing({ nama: "", tingkatId: undefined, jurusanId: null, tahunAjaranId: aktifTa ? Number(aktifTa) : undefined, walikelasId: null, ruangan: "", kapasitas: null });
   280→    setDialogOpen(true);
   281→  };
   282→  const handleEdit = (k: Kelas) => { setEditing({ ...k }); setDialogOpen(true); };
   283→
   284→  const handleSave = async () => {
   285→    if (!editing?.nama?.trim()) { toast({ title: "Nama wajib", variant: "destructive" }); return; }
   286→    if (!editing.tingkatId) { toast({ title: "Tingkat wajib", variant: "destructive" }); return; }
   287→    if (!editing.tahunAjaranId) { toast({ title: "Tahun ajaran wajib", variant: "destructive" }); return; }
   288→    setSaving(true);
   289→    try {
   290→      const url = editing.id ? `/api/kelas/${editing.id}` : "/api/kelas";
   291→      const method = editing.id ? "PUT" : "POST";
   292→      const r = await fetch(url, { method, headers: { "Content-Type": "application/json" }, body: JSON.stringify(editing) });
   293→      const d = await r.json();
   294→      if (!r.ok) throw new Error(d.error || "Gagal menyimpan");
   295→      toast({ title: editing.id ? "Kelas diperbarui" : "Kelas ditambahkan" });
   296→      setDialogOpen(false); setEditing(null); await load();
   297→    } catch (e) {
   298→      toast({ title: "Gagal menyimpan", description: e instanceof Error ? e.message : "", variant: "destructive" });
   299→    } finally { setSaving(false); }
   300→  };
   301→
   302→  const handleDelete = async () => {
   303→    if (!delTarget) return;
   304→    try {
   305→      const r = await fetch(`/api/kelas/${delTarget.id}`, { method: "DELETE" });
   306→      const d = await r.json();
   307→      if (!r.ok) throw new Error(d.error || "Gagal menghapus");
   308→      toast({ title: "Kelas dihapus" });
   309→      setDelTarget(null); await load();
   310→    } catch (e) {
   311→      toast({ title: "Gagal menghapus", description: e instanceof Error ? e.message : "", variant: "destructive" });
   312→    }
   313→  };
   314→
   315→  const update = (k: keyof Kelas, v: unknown) => setEditing((p) => p ? { ...p, [k]: v } : p);
   316→
   317→  return (
   318→    <Card className="border-slate-200">
   319→      <CardContent className="p-4 space-y-4">
   320→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   321→          <div>
   322→            <h3 className="text-base font-semibold text-slate-800">Kelas</h3>
   323→            <p className="text-xs text-slate-500">Kelas per tahun ajaran & tingkat</p>
   324→          </div>
   325→          <div className="flex gap-2">
   326→            <div className="relative w-full sm:w-56">
   327→              <svg className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}>
   328→                <circle cx="11" cy="11" r="8" /><path d="m21 21-4.3-4.3" />
   329→              </svg>
   330→              <Input placeholder="Cari kelas..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-8 h-9" />
   331→            </div>
   332→            <Button size="sm" onClick={handleAdd} className="bg-slate-700 hover:bg-slate-800"><Plus className="h-4 w-4 mr-1" /> Tambah</Button>
   333→          </div>
   334→        </div>
   335→        {loading ? (
   336→          <div className="flex justify-center py-8"><Loader2 className="h-5 w-5 animate-spin text-slate-500" /></div>
   337→        ) : filtered.length === 0 ? (
   338→          <div className="text-center py-8 text-slate-500"><p className="text-sm">Belum ada kelas.</p></div>
   339→        ) : (
   340→          <div className="border border-slate-200 rounded-lg overflow-x-auto">
   341→            <table className="w-full text-sm">
   342→              <thead className="bg-slate-50 text-slate-600">
   343→                <tr>
   344→                  <th className="text-left px-3 py-2 font-medium">Nama</th>
   345→                  <th className="text-left px-3 py-2 font-medium">Tingkat</th>
   346→                  <th className="text-left px-3 py-2 font-medium">Jurusan</th>
   347→                  <th className="text-left px-3 py-2 font-medium">TA</th>
   348→                  <th className="text-left px-3 py-2 font-medium">Walikelas</th>
   349→                  <th className="text-left px-3 py-2 font-medium">Siswa</th>
   350→                  <th className="text-right px-3 py-2 font-medium w-32">Aksi</th>
   351→                </tr>
   352→              </thead>
   353→              <tbody className="divide-y divide-slate-100">
   354→                {filtered.map((k) => (
   355→                  <tr key={k.id} className="hover:bg-slate-50/60">
   356→                    <td className="px-3 py-2 font-medium text-slate-800">{k.nama}{k.ruangan && <span className="text-[10px] text-slate-500 block">{k.ruangan}</span>}</td>
   357→                    <td className="px-3 py-2">{k.tingkat?.nama || "-"}</td>
   358→                    <td className="px-3 py-2">{k.jurusan?.nama || "-"}</td>
   359→                    <td className="px-3 py-2">{k.tahunAjaran?.nama}{k.tahunAjaran?.statusAktif && <Badge className="ml-1 bg-emerald-100 text-emerald-700 text-[10px]">Aktif</Badge>}</td>
   360→                    <td className="px-3 py-2">{k.walikelas?.nama || "-"}</td>
   361→                    <td className="px-3 py-2"><Badge variant="outline" className="text-[10px]">{k._count?.kelasSiswas ?? 0} siswa</Badge></td>
   362→                    <td className="px-3 py-2 text-right whitespace-nowrap">
   363→                      <Button size="sm" variant="outline" className="h-7" onClick={() => setSiswaDialogKelas(k)}>
   364→                        <Users className="h-3 w-3 mr-1" /> Siswa
   365→                      </Button>
   366→                      <Button size="icon" variant="ghost" className="h-7 w-7" onClick={() => handleEdit(k)} title="Edit">
   367→                        <Pencil className="h-3.5 w-3.5" />

   484→function KelasSiswaManager({ kelas }: { kelas: Kelas }) {
   485→  const [list, setList] = useState<KelasSiswa[]>([]);
   486→  const [allSiswa, setAllSiswa] = useState<SiswaMini[]>([]);
   487→  const [loading, setLoading] = useState(true);
   488→  const [pick, setPick] = useState<string>("");
   489→  const [adding, setAdding] = useState(false);
   490→  const { toast } = useToast();
   491→
   492→  const load = useCallback(async () => {
   493→    setLoading(true);
   494→    try {
   495→      const [r1, r2] = await Promise.all([
   496→        fetch(`/api/kelas-siswa?kelasId=${kelas.id}`),
   497→        fetch("/api/siswa"),
   498→      ]);
   499→      const d1 = await r1.json();
   500→      const d2 = await r2.json();
   501→      if (Array.isArray(d1)) setList(d1);
   502→      if (Array.isArray(d2)) setAllSiswa(d2);
   503→    } catch { /* ignore */ } finally { setLoading(false); }
   504→  }, [kelas.id]);
   505→
   506→  useEffect(() => { load(); }, [load]);
   507→
   508→  const linkedIds = new Set(list.map((x) => x.siswaId));
   509→  const available = allSiswa.filter((s) => !linkedIds.has(s.id));
   510→
   511→  const handleAdd = async () => {
   512→    if (!pick) return;
   513→    setAdding(true);
   514→    try {
   515→      const r = await fetch("/api/kelas-siswa", {
   516→        method: "POST", headers: { "Content-Type": "application/json" },
   517→        body: JSON.stringify({ kelasId: kelas.id, siswaId: Number(pick), tahunAjaranId: kelas.tahunAjaranId }),
   518→      });
   519→      const d = await r.json();
   520→      if (!r.ok) throw new Error(d.error || "Gagal menambah");
   521→      setPick("");
   522→      await load();
   523→      toast({ title: "Siswa ditambahkan ke kelas" });
   524→    } catch (e) {
   525→      toast({ title: "Gagal menambah", description: e instanceof Error ? e.message : "", variant: "destructive" });
   526→    } finally { setAdding(false); }
   527→  };
   528→
   529→  const handleRemove = async (ksid: number, nama: string) => {
   530→    try {
   531→      const r = await fetch(`/api/kelas-siswa/${ksid}`, { method: "DELETE" });
   532→      if (!r.ok) throw new Error("Gagal menghapus");
   533→      await load();
   534→      toast({ title: `${nama} dihapus dari kelas` });
   535→    } catch (e) {
   536→      toast({ title: "Gagal menghapus", description: e instanceof Error ? e.message : "", variant: "destructive" });
   537→    }
   538→  };
   539→
   540→  return (
   541→    <div className="space-y-3">
   542→      {loading ? (
   543→        <div className="flex justify-center py-6"><Loader2 className="h-5 w-5 animate-spin text-slate-500" /></div>
   544→      ) : (
   545→        <>
   546→          <div className="flex flex-col sm:flex-row gap-2 items-end">
   547→            <div className="flex-1 w-full">
   548→              <Label className="text-xs">Tambah Siswa ke Kelas</Label>
   549→              <Select value={pick} onValueChange={setPick}>
   550→                <SelectTrigger className="w-full"><SelectValue placeholder="Pilih siswa..." /></SelectTrigger>
   551→                <SelectContent>
   552→                  {available.length === 0 ? (
   553→                    <SelectItem value="_none" disabled>Tidak ada siswa tersedia</SelectItem>
   554→                  ) : available.map((s) => (
   555→                    <SelectItem key={s.id} value={String(s.id)}>{s.nama} {s.nis ? `(${s.nis})` : ""}</SelectItem>
   556→                  ))}
   557→                </SelectContent>
   558→              </Select>
   559→            </div>
   560→            <Button onClick={handleAdd} disabled={adding || !pick || pick === "_none"} className="bg-slate-700 hover:bg-slate-800">
   561→              {adding ? <Loader2 className="h-4 w-4 animate-spin" /> : <Plus className="h-4 w-4 mr-1" />} Tambah
   562→            </Button>
   563→          </div>
   564→          <div className="border border-slate-200 rounded-lg overflow-hidden max-h-72 overflow-y-auto">
   565→            {list.length === 0 ? (
   566→              <div className="text-center py-6 text-sm text-slate-500">Belum ada siswa di kelas ini.</div>
   567→            ) : (
   568→              <table className="w-full text-sm">
   569→                <thead className="bg-slate-50 text-slate-600 sticky top-0">
   570→                  <tr>
   571→                    <th className="text-left px-3 py-2 font-medium">Nama</th>
